import { Children, Fragment, useEffect, useRef, useState } from 'react';
import {
  Box,
  Button,
  Collapse,
  Dialog,
  DialogTitle,
  DialogContent,
  DialogActions,
  Divider,
  IconButton,
  Tooltip,
  MenuItem,
  Radio,
  RadioGroup,
  FormControlLabel,
  Select,
  Slider,
  Switch,
  TextField,
  ToggleButton,
  ToggleButtonGroup,
  Typography,
} from '@mui/material';
import { createAppearanceSemanticTokens } from '@/theme/tokens/semantic';
import { useTheme } from '@mui/material/styles';
import { CopyIcon, DownloadIcon, ExpandMoreIcon } from '@/components/icons';
import { useAppearanceReducedMotion } from '@/hooks/useAppearanceReducedMotion';
import { useSettings } from '@/contexts/SettingsContext';
import { PreferenceRow } from '@/features/overlays/preference-surface';
import {
  ACCENTS,
  DEFAULT_APPEARANCE,
  updateProfile,
  selectPreset,
  importTheme,
  exportTheme,
} from '@/theme/appearance';
import { ColorControl, NumberControl } from './AppearanceControls';
function Row(props) {
  return <PreferenceRow variant="appearance" tabIndex={-1} {...props} />;
}
function Group({ label, id, children }) {
  return (
    <Box
      component="section"
      role="group"
      aria-label={label}
      id={id}
      tabIndex={id ? -1 : undefined}
      data-appearance-group
      sx={{
        border: '1px solid',
        borderColor: 'divider',
        borderRadius: '10px',
        overflow: 'visible',
        mb: 1.5,
      }}
    >
      {Children.toArray(children).map((child, i) => (
        <Fragment key={child.key || i}>
          {i > 0 && <Divider sx={{ mx: 1.5 }} />}
          {child}
        </Fragment>
      ))}
    </Box>
  );
}
const selectSx = {
  height: 30,
  borderRadius: '999px',
  fontSize: 12,
  minWidth: 112,
  '& .MuiSelect-select': { py: 0.5, px: 1.25 },
  '& .MuiSelect-icon': { fontSize: 18 },
  backgroundColor: 'transparent',
};
export default function AppearanceSection({ active = true, ...props }) {
  const { settingsScope } = useSettings();
  return active ? <AppearancePanel key={settingsScope} {...props} /> : null;
}
function AppearancePanel({ revealTarget, revealRequest, onTargetReady }) {
  const { settings, updateSetting } = useSettings(),
    theme = useTheme(),
    persisted = settings.appearance,
    effective = theme.palette.mode;
  const reducedMotion = useAppearanceReducedMotion();
  const [advanced, setAdvanced] = useState(false),
    [importMode, setImportMode] = useState(null),
    [share, setShare] = useState(''),
    [importError, setImportError] = useState(''),
    [status, setStatus] = useState('');
  const [interaction, setInteraction] = useState(null);
  const cancelledGesture = useRef(false);
  const a = interaction
    ? {
        ...persisted,
        profiles: {
          ...persisted.profiles,
          [interaction.mode]: {
            ...persisted.profiles[interaction.mode],
            ...interaction.patch,
            custom: true,
          },
        },
      }
    : persisted;
  const previewTokens = interaction ? createAppearanceSemanticTokens(a, effective) : null;
  const preview = (mode, patch) => {
    if (Object.entries(patch).every(([key, value]) => persisted.profiles[mode][key] === value)) {
      setInteraction(null);
      return;
    }
    setInteraction({ mode, patch });
  };
  const cancelPreview = () => setInteraction(null);
  useEffect(() => {
    if (revealTarget && /font|motion|separate|contrast|full-width|diff/.test(revealTarget)) {
      // eslint-disable-next-line react-hooks/set-state-in-effect -- Search reveals a collapsed target without changing preferences.
      setAdvanced(true);
    }
  }, [revealTarget, revealRequest]);
  const save = (patch) => updateSetting('appearance', { ...a, ...patch });
  const edit = (mode, patch) => {
    const next = updateProfile(persisted, mode, patch);
    if (JSON.stringify(next) !== JSON.stringify(persisted)) updateSetting('appearance', next);
    setInteraction(null);
  };
  const modes = a.separateModes ? ['light', 'dark'] : [effective];
  const segmented = (label, value, options, onChange) => (
    <ToggleButtonGroup
      size="small"
      exclusive
      aria-label={label}
      value={value}
      onChange={(_, v) => {
        if (v !== null) onChange(v);
      }}
      sx={{
        border: 0,
        p: 0.25,
        gap: 0.25,
        backgroundColor: 'background.sunken',
        borderRadius: '999px',
        '& .MuiToggleButton-root': {
          minHeight: 26,
          height: 26,
          border: '0 !important',
          borderRadius: '999px !important',
          px: 1.25,
          fontSize: 12,
          whiteSpace: 'nowrap',
        },
        '& .Mui-selected': { backgroundColor: 'action.selected' },
      }}
    >
      {options.map(([id, text]) => (
        <ToggleButton key={id} value={id}>
          {text}
        </ToggleButton>
      ))}
    </ToggleButtonGroup>
  );
  const toggle = (label, key, id, description) => (
    <Row label={label} id={id} description={description}>
      <Switch
        size="small"
        checked={a[key]}
        onChange={(e) => save({ [key]: e.target.checked })}
        slotProps={{ input: { role: 'switch', 'aria-label': label } }}
      />
    </Row>
  );
  return (
    <Box
      data-appearance-panel
      onKeyDown={(event) => {
        if (event.key === 'Escape' && interaction) {
          event.preventDefault();
          event.stopPropagation();
          cancelledGesture.current = true;
          setInteraction(null);
        }
      }}
      style={
        previewTokens
          ? {
              '--appearance-preview-bg': previewTokens.background.default,
              '--appearance-preview-ink': previewTokens.text.primary,
              '--appearance-preview-secondary': previewTokens.text.secondary,
              '--appearance-preview-border': previewTokens.border.idle,
            }
          : undefined
      }
      sx={(theme) => ({
        '& [data-appearance-group]': {
          backgroundColor: 'var(--appearance-preview-bg, transparent)',
          borderColor: `var(--appearance-preview-border, ${theme.palette.divider})`,
        },
        '& [data-appearance-group] .MuiTypography-root, & [data-appearance-group] input': {
          color: `var(--appearance-preview-ink, ${theme.palette.text.primary})`,
        },
        '& [data-appearance-group] [data-appearance-description], & [data-appearance-group] [data-appearance-unit]':
          {
            color: `var(--appearance-preview-secondary, ${theme.palette.text.secondary})`,
          },
      })}
    >
      <Typography
        component="h2"
        sx={{ fontSize: 16, fontWeight: 500, lineHeight: '24px', mb: 1.5 }}
      >
        Visual style
      </Typography>
      <Group label="Mode">
        <Row label="Mode" id="appearance-mode" sx={{ minHeight: 76 }}>
          <RadioGroup
            row
            aria-label="Mode"
            value={a.mode}
            onChange={(e) => save({ mode: e.target.value })}
            sx={{ gap: { xs: 1, sm: 2 }, flexWrap: 'nowrap' }}
          >
            {['system', 'light', 'dark'].map((mode) => (
              <FormControlLabel
                key={mode}
                value={mode}
                control={
                  <Radio
                    size="small"
                    slotProps={{ input: { 'aria-label': mode[0].toUpperCase() + mode.slice(1) } }}
                    sx={{ position: 'absolute', opacity: 0 }}
                  />
                }
                label={
                  <Box
                    aria-hidden="true"
                    sx={{
                      width: { xs: 72, sm: 80 },
                      aspectRatio: '4 / 3',
                      borderRadius: '7px',
                      overflow: 'hidden',
                      background:
                        mode === 'system'
                          ? 'linear-gradient(90deg,#fcfcfc 50%,#212121 50%)'
                          : mode === 'dark'
                            ? '#212121'
                            : '#fcfcfc',
                      p: 1,
                      display: 'flex',
                      gap: 0.75,
                    }}
                  >
                    <Box
                      sx={{
                        width: 12,
                        borderRadius: 0.5,
                        background: mode === 'dark' ? '#414141' : '#cdcdcd',
                      }}
                    />
                    <Box sx={{ flex: 1, mt: 0.75 }}>
                      {[1, 2, 3].map((n) => (
                        <Box
                          key={n}
                          sx={{
                            height: 3,
                            mb: 0.75,
                            borderRadius: 1,
                            background: mode === 'dark' ? '#afafaf' : '#8f8f8f',
                            width: n === 3 ? '65%' : '100%',
                          }}
                        />
                      ))}
                    </Box>
                  </Box>
                }
                sx={{
                  m: 0,
                  position: 'relative',
                  borderRadius: '8px',
                  outline: a.mode === mode ? '2px solid' : '1px solid',
                  outlineColor: a.mode === mode ? 'text.secondary' : 'divider',
                  outlineOffset: 2,
                  '&:focus-within': {
                    outline: '2px solid',
                    outlineColor: 'accent.foreground',
                    outlineOffset: 3,
                  },
                  '& .MuiFormControlLabel-label': { lineHeight: 0 },
                }}
              />
            ))}
          </RadioGroup>
        </Row>
      </Group>
      {modes.map((mode) => {
        const p = a.profiles[mode];
        return (
          <Group key={mode} label={a.separateModes ? `${mode} theme settings` : 'Theme settings'}>
            <Box
              id={mode === effective ? 'appearance-theme' : undefined}
              tabIndex={-1}
              sx={{
                display: 'flex',
                alignItems: 'center',
                minHeight: 40,
                px: 1.5,
                py: 0.5,
                gap: 1,
              }}
            >
              <Typography component="h3" sx={{ fontSize: 14, fontWeight: 500, mr: 'auto' }}>
                {a.separateModes ? `${mode === 'light' ? 'Light' : 'Dark'} theme` : 'Theme'}
              </Typography>
              <Tooltip title="Import theme">
                <IconButton
                  size="small"
                  aria-label={a.separateModes ? `Import ${mode} theme` : 'Import theme'}
                  onClick={() => {
                    setImportMode(mode);
                    setShare('');
                    setImportError('');
                  }}
                  sx={{ width: 28, height: 28, minWidth: 28 }}
                >
                  <DownloadIcon sx={{ fontSize: 16, transform: 'rotate(180deg)' }} />
                </IconButton>
              </Tooltip>
              <Tooltip title="Copy theme">
                <IconButton
                  size="small"
                  aria-label={a.separateModes ? `Copy ${mode} theme` : 'Copy theme'}
                  onClick={async () => {
                    try {
                      await navigator.clipboard.writeText(exportTheme(p, mode));
                      setStatus('Theme copied.');
                    } catch {
                      setStatus('Could not copy theme. Check clipboard permissions and try again.');
                    }
                  }}
                  sx={{ width: 28, height: 28, minWidth: 28 }}
                >
                  <CopyIcon sx={{ fontSize: 16 }} />
                </IconButton>
              </Tooltip>
              <Select
                size="small"
                value={p.presetId}
                inputProps={{ 'aria-label': `${mode} theme preset` }}
                sx={selectSx}
              >
                {[
                  ['chatgpt', 'ChatGPT'],
                  ['moonlit', 'Moonlit'],
                ].map(([id, label]) => (
                  <MenuItem
                    key={id}
                    value={id}
                    onClick={() => updateSetting('appearance', selectPreset(a, mode, id))}
                  >
                    Aa · {label}
                  </MenuItem>
                ))}
              </Select>
            </Box>
            <Row label="Accent" id={mode === effective ? 'appearance-accent' : undefined}>
              <Select
                size="small"
                value={p.accent}
                inputProps={{ 'aria-label': `${mode} accent` }}
                onChange={(e) => edit(mode, { accent: e.target.value })}
                sx={selectSx}
              >
                {ACCENTS.map((c) => (
                  <MenuItem key={c.id} value={c.id}>
                    <Box
                      component="span"
                      sx={{
                        width: 12,
                        height: 12,
                        borderRadius: '50%',
                        mr: 1,
                        display: 'inline-block',
                        backgroundColor: c.swatch || p.customAccent,
                        border: '1px solid',
                        borderColor: 'divider',
                      }}
                    />
                    {c.label}
                    {p.accent === c.id ? ' ✓' : ''}
                  </MenuItem>
                ))}
              </Select>
            </Row>
            {p.accent === 'custom' && (
              <Row label="Custom accent">
                <ColorControl
                  label="Custom accent"
                  value={persisted.profiles[mode].customAccent}
                  onChange={(color) => edit(mode, { customAccent: color })}
                  onPreview={(color) => preview(mode, { customAccent: color })}
                  onCancelPreview={cancelPreview}
                />
              </Row>
            )}
            <Row label="Background" id={mode === effective ? 'appearance-background' : undefined}>
              <ColorControl
                label={a.separateModes ? `${mode} background` : 'Background'}
                value={persisted.profiles[mode].surface}
                onChange={(color) => edit(mode, { surface: color })}
                onPreview={(color) => preview(mode, { surface: color })}
                onCancelPreview={cancelPreview}
              />
            </Row>
            <Row label="Foreground" id={mode === effective ? 'appearance-foreground' : undefined}>
              <ColorControl
                label={a.separateModes ? `${mode} foreground` : 'Foreground'}
                value={persisted.profiles[mode].ink}
                onChange={(color) => edit(mode, { ink: color })}
                onPreview={(color) => preview(mode, { ink: color })}
                onCancelPreview={cancelPreview}
              />
            </Row>
          </Group>
        );
      })}
      <Box
        sx={{
          display: 'flex',
          alignItems: 'center',
          justifyContent: 'space-between',
          mt: 3,
          mb: 1.25,
        }}
      >
        <Button
          aria-expanded={advanced}
          aria-controls="appearance-advanced"
          onClick={() => setAdvanced((v) => !v)}
          endIcon={<ExpandMoreIcon sx={{ transform: advanced ? 'rotate(180deg)' : 'none' }} />}
          sx={{ p: 0, minHeight: 32, fontSize: 14, fontWeight: 500 }}
        >
          Advanced
        </Button>
        {JSON.stringify(a) !== JSON.stringify(DEFAULT_APPEARANCE) && (
          <Button
            size="small"
            onClick={() => updateSetting('appearance', DEFAULT_APPEARANCE)}
            sx={{ p: 0, fontSize: 12, color: 'text.secondary' }}
          >
            Reset to default
          </Button>
        )}
      </Box>
      <Collapse
        in={advanced}
        timeout={reducedMotion ? 0 : 'auto'}
        unmountOnExit
        onEntered={() => onTargetReady?.()}
      >
        <Box id="appearance-advanced">
          <Group label="Font sizes" id="appearance-font">
            {[
              ['uiFontSize', 'UI font size', 11, 16, 'Base size for app navigation and controls'],
              [
                'contentFontSize',
                'Content font size',
                12,
                24,
                'Text size in responses and reading content',
              ],
              ['codeFontSize', 'Code font size', 8, 24, 'Text size in code blocks and SQL editors'],
            ].map(([key, label, min, max, description]) => (
              <Row key={key} label={label} description={description}>
                <NumberControl
                  label={label}
                  value={a[key]}
                  min={min}
                  max={max}
                  onChange={(value) => save({ [key]: value })}
                />
              </Row>
            ))}
          </Group>
          <Group label="Motion and mode profiles">
            <Row
              label="Reduce motion"
              description="Control interface animations"
              id="appearance-reduced-motion"
            >
              {segmented(
                'Reduce motion',
                a.reducedMotion,
                [
                  ['system', 'System'],
                  ['on', 'On'],
                  ['off', 'Off'],
                ],
                (v) => save({ reducedMotion: v }),
              )}
            </Row>
            {toggle(
              'Separate light and dark modes',
              'separateModes',
              'appearance-separate-modes',
              'Customize each mode independently',
            )}
          </Group>
          {modes.map((mode) => (
            <Group key={mode} label={`${mode} contrast settings`}>
              <Row
                label={
                  a.separateModes ? `${mode === 'light' ? 'Light' : 'Dark'} contrast` : 'Contrast'
                }
                id={mode === effective ? 'appearance-contrast' : undefined}
              >
                <Box
                  sx={{
                    width: 180,
                    maxWidth: '100%',
                    display: 'flex',
                    gap: 2,
                    alignItems: 'center',
                  }}
                >
                  <Slider
                    min={0}
                    max={100}
                    step={1}
                    value={a.profiles[mode].contrast}
                    aria-label={`${mode === 'light' ? 'Light' : 'Dark'} contrast`}
                    onMouseDownCapture={() => {
                      cancelledGesture.current = false;
                    }}
                    onTouchStartCapture={() => {
                      cancelledGesture.current = false;
                    }}
                    onChange={(_, v) => {
                      if (!cancelledGesture.current) preview(mode, { contrast: v });
                    }}
                    onChangeCommitted={(_, v) => {
                      if (cancelledGesture.current) {
                        cancelledGesture.current = false;
                        cancelPreview();
                        return;
                      }
                      edit(mode, { contrast: v });
                    }}
                    onBlur={() => {
                      if (
                        !cancelledGesture.current &&
                        interaction?.mode === mode &&
                        'contrast' in interaction.patch
                      )
                        edit(mode, interaction.patch);
                    }}
                    sx={{ height: 2, py: 1, '& .MuiSlider-thumb': { width: 12, height: 12 } }}
                  />
                  <output style={{ fontSize: 12, minWidth: 20, textAlign: 'right' }}>
                    {a.profiles[mode].contrast}
                  </output>
                </Box>
              </Row>
            </Group>
          ))}
          <Group label="Diff settings">
            <Row
              label="Diff markers"
              description="Identify added and deleted lines"
              id="appearance-diff-markers"
            >
              {segmented(
                'Diff markers',
                a.diffMarkers,
                [
                  ['color', 'Color'],
                  ['sign', '+/-'],
                ],
                (v) => save({ diffMarkers: v }),
              )}
            </Row>
          </Group>
        </Box>
      </Collapse>
      {status && (
        <Typography
          role={status === 'Theme copied.' ? 'status' : 'alert'}
          sx={{ mt: 2, fontSize: 12 }}
        >
          {status}
        </Typography>
      )}
      <Dialog
        open={!!importMode}
        onClose={() => setImportMode(null)}
        fullWidth
        maxWidth="sm"
        aria-labelledby="import-theme-title"
      >
        <DialogTitle id="import-theme-title">Import theme</DialogTitle>
        <DialogContent>
          <TextField
            autoFocus
            fullWidth
            multiline
            minRows={4}
            label="Theme share string"
            value={share}
            onChange={(e) => setShare(e.target.value)}
            slotProps={{ htmlInput: { maxLength: 20000 } }}
            sx={{ mt: 1 }}
          />
          <Typography sx={{ mt: 1, fontSize: 12, color: 'text.secondary' }}>
            Supports Moonlit v1 and validated Codex v1 colors, contrast and supported fonts. Codex
            syntax maps to Moonlit.
          </Typography>
          {importError && (
            <Typography role="alert" color="error">
              {importError}
            </Typography>
          )}
        </DialogContent>
        <DialogActions>
          <Button onClick={() => setImportMode(null)}>Cancel</Button>
          <Button
            disabled={!share.trim()}
            onClick={() => {
              try {
                const result = importTheme(share);
                save({ profiles: { ...a.profiles, [result.mode]: result.profile } });
                setImportMode(null);
              } catch (e) {
                setImportError(e.message);
              }
            }}
          >
            Import
          </Button>
        </DialogActions>
      </Dialog>
    </Box>
  );
}
