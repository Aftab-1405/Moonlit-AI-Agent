import React, { Profiler, useEffect, useRef, useState } from 'react';
import { createRoot } from 'react-dom/client';
import { CssBaseline } from '@mui/material';
import { useTheme } from '@mui/material/styles';
import { BrowserRouter } from 'react-router-dom';
import { ThemeProvider } from '@/contexts/ThemeContext';
import { useSettings } from '@/contexts/SettingsContext';
import SettingsModal from '@/features/overlays/settings/SettingsModal';
import ChatInput from '@/features/chat/ChatInput';
import SqlEditorSurface from '@/features/sidebar-right/artifacts/sql-workspace/SqlEditorSurface';
import PerspectiveDashboard from '@/features/sidebar-right/artifacts/data-visualization/PerspectiveDashboard';
import { DEFAULT_APPEARANCE } from '@/theme/appearance';

window.perfActive = false;
window.resetPerf = () => { window.perfMetrics = { themes: 0, writes: 0, boot: 0, commits: [], frames: [], longTasks: [], panelRenders: 0 }; window.perfActive = true; };
window.resetPerf(); window.perfActive = false;
const originalSet = Storage.prototype.setItem;
Storage.prototype.setItem = function(key, value) { if(window.perfActive) window.perfMetrics[key === 'moonlit-appearance-boot-v1' ? 'boot' : 'writes']++; return originalSet.call(this,key,value); };
new PerformanceObserver(list => { if(window.perfActive) window.perfMetrics.longTasks.push(...list.getEntries().map(e=>e.duration)); }).observe({type:'longtask',buffered:false});
let last = performance.now();
function frame(now) { if(window.perfActive) window.perfMetrics.frames.push(now-last); last=now; requestAnimationFrame(frame); } requestAnimationFrame(frame);
const data = { columns: ['region','sales'], rows: [['North',42],['South',32],['West',28]] };
function Fixture() {
  const { settings, updateSetting, settingsScope } = useSettings();
  const theme = useTheme(), previous = useRef(theme);
  const [query,setQuery] = useState('SELECT region, sales FROM sales;');
  useEffect(() => { if (previous.current !== theme && window.perfActive) window.perfMetrics.themes++; previous.current=theme; },[theme]);
  window.qa = { settings, settingsScope, setAppearance: value=>updateSetting('appearance',value), defaults: DEFAULT_APPEARANCE, theme: theme.palette };
  return <><CssBaseline/><div style={{position:'fixed',width:'min(800px, 100vw)',height:500,visibility:'hidden',overflow:'hidden'}}><ChatInput onSend={()=>{}}/><SqlEditorSurface query={query} onQueryChange={setQuery}/><PerspectiveDashboard data={data} storageKey="interaction-qa-chart"/></div><SettingsModal open onClose={()=>{}} initialSection="appearance"/></>;
}
createRoot(document.getElementById('root')).render(<Profiler id="fixture" onRender={(_,phase,time)=>{if(window.perfActive)window.perfMetrics.commits.push({phase,time});}}><BrowserRouter><ThemeProvider><Fixture/></ThemeProvider></BrowserRouter></Profiler>);
