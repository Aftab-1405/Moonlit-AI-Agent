import { useCallback, useEffect, useRef, useState } from 'react';
import { Box, Button, Popover, Slider, TextField, Typography } from '@mui/material';
import { normalizeHex } from '@/theme/appearance';
import { hexToHsl, hslToHex } from '@/theme/colorModel';
export function NumberControl({ label, value, min, max, onChange }) {
  const [draft, setDraft] = useState(String(value)),
    [error, setError] = useState(false),
    lastCommit = useRef(value);
  useEffect(() => {
    // eslint-disable-next-line react-hooks/set-state-in-effect -- Reconcile reset/account changes without persisting drafts.
    setDraft(String(value));
    setError(false);
    lastCommit.current = value;
  }, [value]);
  const commit = () => {
    const n = Number(draft);
    if (draft.trim() !== '' && Number.isInteger(n) && n >= min && n <= max) {
      if (n !== lastCommit.current) {
        lastCommit.current = n;
        onChange(n);
      }
      setError(false);
    } else setError(true);
  };
  return (
    <Box sx={{ display: 'flex', alignItems: 'center', gap: 0.75 }}>
      <TextField
        type="number"
        size="small"
        value={draft}
        error={error}
        onChange={(e) => setDraft(e.target.value)}
        onBlur={commit}
        onKeyDown={(e) => {
          if (e.key === 'Enter') {
            e.preventDefault();
            commit();
          }
          if (e.key === 'Escape') {
            e.stopPropagation();
            e.preventDefault();
            setDraft(String(value));
            setError(false);
          }
        }}
        slotProps={{
          htmlInput: {
            'aria-label': label,
            min,
            max,
            step: 1,
            'aria-description': error ? `Enter a whole number from ${min} to ${max}.` : undefined,
          },
        }}
        sx={{
          width: '64px !important',
          minWidth: 64,
          maxWidth: 64,
          '& .MuiInputBase-root': { height: 28, minHeight: 28 },
          '& input': { p: '4px 8px', fontSize: 14 },
        }}
      />
      <Typography data-appearance-unit sx={{ fontSize: 12, color: 'text.secondary' }}>
        px
      </Typography>
    </Box>
  );
}
export function ColorControl({ label, value, onChange, onPreview, onCancelPreview }) {
  const [draft, setDraft] = useState(value),
    [error, setError] = useState(''),
    [anchor, setAnchor] = useState(null),
    [color, setColor] = useState(() => hexToHsl(value));
  const lastCommit = useRef(value);
  const colorRef = useRef(color);
  useEffect(() => {
    // eslint-disable-next-line react-hooks/set-state-in-effect -- Reconcile profile resets and account changes.
    setDraft(value);
    setError('');
    colorRef.current = hexToHsl(value);
    setColor(colorRef.current);
    lastCommit.current = value;
  }, [value]);
  const commit = () => {
    const hex = normalizeHex(draft);
    if (hex) {
      if (hex !== lastCommit.current) {
        lastCommit.current = hex;
        onChange(hex);
      }
      setDraft(hex);
      setError('');
    } else setError('Enter a valid 6-digit hex color.');
  };
  const choose = (patch) => {
    const next = { ...colorRef.current, ...patch },
      hex = hslToHex(next);
    colorRef.current = next;
    setColor(next);
    setDraft(hex);
    setError('');
    if (onPreview) onPreview(hex);
    else if (hex !== lastCommit.current) {
      lastCommit.current = hex;
      onChange(hex);
    }
  };
  const commitPicker = () => {
    const hex = hslToHex(colorRef.current);
    if (hex !== lastCommit.current) {
      lastCommit.current = hex;
      onChange(hex);
    } else onCancelPreview?.();
  };
  const nativePickerRef = useCallback(
    (node) => {
      if (!node) return;
      const commitNative = () => {
        const hex = normalizeHex(node.value);
        if (!hex) return;
        colorRef.current = hexToHsl(hex);
        setColor(colorRef.current);
        setDraft(hex);
        setError('');
        if (hex !== lastCommit.current) {
          lastCommit.current = hex;
          onChange(hex);
        }
      };
      node.addEventListener('change', commitNative);
      return () => node.removeEventListener('change', commitNative);
    },
    [onChange],
  );
  return (
    <Box sx={{ position: 'relative' }}>
      <Box
        sx={{
          display: 'flex',
          alignItems: 'center',
          gap: 0.5,
          border: '1px solid',
          borderColor: error ? 'error.main' : 'divider',
          borderRadius: '999px',
          px: 0.5,
          height: 30,
          width: 112,
        }}
      >
        <Button
          aria-label={`${label} color`}
          onClick={(e) => {
            colorRef.current = hexToHsl(value);
            setColor(colorRef.current);
            setAnchor(e.currentTarget);
          }}
          sx={{
            minWidth: 28,
            width: 28,
            height: 28,
            p: 0,
            borderRadius: '50%',
            flexShrink: 0,
            '&::after': {
              content: '""',
              width: 16,
              height: 16,
              borderRadius: '50%',
              border: '1px solid',
              borderColor: 'divider',
              backgroundColor: normalizeHex(draft) || value,
            },
          }}
        />
        <Box
          component="input"
          aria-label={label}
          aria-invalid={!!error}
          aria-describedby={error ? `${label.replace(/\s/g, '-')}-error` : undefined}
          value={draft}
          maxLength={7}
          onChange={(e) => setDraft(e.target.value)}
          onBlur={commit}
          onKeyDown={(e) => {
            if (e.key === 'Enter') {
              e.preventDefault();
              commit();
            }
            if (e.key === 'Escape') {
              e.preventDefault();
              e.stopPropagation();
              setDraft(value);
              setError('');
            }
          }}
          sx={{
            width: 70,
            minWidth: 0,
            border: 0,
            p: 0,
            outline: 0,
            background: 'transparent',
            color: 'text.primary',
            fontFamily: 'inherit',
            fontSize: '12px',
            '&:focus-visible': {
              outline: '2px solid',
              outlineColor: 'text.secondary',
              outlineOffset: 1,
            },
          }}
        />
      </Box>
      {error && (
        <Typography
          id={`${label.replace(/\s/g, '-')}-error`}
          role="alert"
          sx={{
            position: 'absolute',
            top: 32,
            right: 0,
            width: 180,
            fontSize: 11,
            color: 'error.main',
            zIndex: 1,
            backgroundColor: 'background.paper',
          }}
        >
          {error}
        </Typography>
      )}
      <Popover
        open={!!anchor}
        anchorEl={anchor}
        onClose={(_, reason) => {
          if (reason === 'escapeKeyDown') {
            setDraft(value);
            colorRef.current = hexToHsl(value);
            setColor(colorRef.current);
            onCancelPreview?.();
          } else commitPicker();
          setAnchor(null);
        }}
        anchorOrigin={{ vertical: 'bottom', horizontal: 'left' }}
      >
        <Box sx={{ width: 230, p: 2 }}>
          <Box
            component="input"
            type="color"
            aria-label="Choose color"
            value={hslToHex(color)}
            ref={nativePickerRef}
            onInput={(event) => choose(hexToHsl(event.currentTarget.value))}
            // React also emits onChange for continuous input; the native change listener commits.
            onChange={() => {}}
            onBlur={commitPicker}
            sx={{ width: '100%', height: 56, border: 0, backgroundColor: 'transparent' }}
          />
          <Typography>Color</Typography>
          <Slider
            aria-label="Color"
            value={color.l * 100}
            min={0}
            max={100}
            onChange={(_, v) => choose({ l: v / 100 })}
            onChangeCommitted={commitPicker}
            onBlur={commitPicker}
          />
          <Typography>Hue</Typography>
          <Slider
            aria-label="Hue"
            value={color.h}
            min={0}
            max={359}
            onChange={(_, v) => choose({ h: v })}
            onChangeCommitted={commitPicker}
            onBlur={commitPicker}
          />
        </Box>
      </Popover>
    </Box>
  );
}
